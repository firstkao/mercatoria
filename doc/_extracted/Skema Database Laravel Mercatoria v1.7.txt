<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Store extends Model
{
 protected $fillable = [
 'name',
 'min_belanja_rmb',
 'ongkir_rmb',
 ];

 protected $casts = [
 'min_belanja_rmb' => 'decimal:2',
 'ongkir_rmb' => 'decimal:2',
 ];

 // ─── Relationships ───────────────────────────────────

 public function games(): HasMany
 {
 return $this->hasMany(Game::class, 'default_store_id');
 }

 public function products(): HasMany
 {
 return $this->hasMany(Product::class);
 }

 // ─── Helpers ─────────────────────────────────────────

 /**
 * Cek apakah store gratis selalu (ongkir = 0)
 */
 public function isFreeShipping(): bool
 {
 return $this->ongkir_rmb == 0;
 }

 /**
 * Cek apakah min_belanja aktif (bukan 0)
 */
 public function hasMinBelanja(): bool
 {
 return $this->min_belanja_rmb > 0;
 }
}

10. Service BARU: ShippingCalculatorService (Per-Varian RMB & Single-Profit) UPD v1.4

app/Services/Order/ShippingCalculatorService.php

<?php

namespace App\Services\Order;

use App\Models\Product;
use App\Models\ProductVariant;

class ShippingCalculatorService
{
 /**
 * Hitung ongkir lokal CN (per varian / produk) dalam IDR
 * v1.4: Menggunakan harga RMB spesifik varian ($variant->effectivePriceRmb())
 */
 public function calculateLocalCn(
 Product $product,
 ?ProductVariant $variant = null
 ): float {
 $store = $product->effectiveStore();

 // Kalau gak ada store → gratis
 if (! $store) {
 return 0;
 }

 // Kalau ongkir = 0 → gratis selalu
 if ($store->isFreeShipping()) {
 return 0;
 }

 // Kalau min_belanja = 0 → gratis (biar gak divide by zero)
 if (! $store->hasMinBelanja()) {
 return 0;
 }

 // v1.4: Ambil harga RMB dari varian terlebih dahulu, fallback ke produk induk
 $hargaRmb = $variant
 ? $variant->effectivePriceRmb()
 : (float) $product->price_rmb;

 $minBelanja = (float) $store->min_belanja_rmb;
 $ongkirRmb = (float) $store->ongkir_rmb;

 // Kalau harga ≥ min belanja → gratis
 if ($hargaRmb >= $minBelanja) {
 return 0;
 }

 // Hitung proporsional: (harga_rmb × ongkir_rmb) / min_belanja_rmb
 $ongkirLokalRmb = ($hargaRmb * $ongkirRmb) / $minBelanja;
 $rate = (float) setting('rate_rmb_idr', 3000);

 return $ongkirLokalRmb * $rate;
 }

 /**
 * Hitung ongkir CN-BTM dalam IDR berdasarkan berat
 */
 public function calculateCnBtm(int $weightGram): float
 {
 $rate = (float) setting('ongkir_cn_btm_per_kg', 25000);
 $beratKg = $weightGram / 1000;

 return $beratKg * $rate;
 }

 /**
 * Hitung ongkir BTM-JKT dalam IDR berdasarkan berat
 */
 public function calculateBtmJkt(int $weightGram): float
 {
 $rate = (float) setting('ongkir_btm_jkt_per_kg', 60000);
 $beratKg = $weightGram / 1000;

 return $beratKg * $rate;
 }

 /**
 * Hitung SEMUA ongkir + Harga Bersih Final untuk 1 item/varian
 * v1.4: Harga Bersih sudah termasuk Profit 11% + CEILING 5.000 per item
 * sehingga harga di Katalog, Detail Produk, Keranjang, dan Checkout 100% SAMA.
 */
 public function calculateItemShipping(
 Product $product,
 ?ProductVariant $variant = null
 ): array {
 $weightGram = $variant?->effectiveWeight() ?? $product->effectiveWeight();
 $hargaRmb = $variant ? $variant->effectivePriceRmb() : (float) $product->price_rmb;
 $rate = (float) setting('rate_rmb_idr', 3000);
 $profitPct = ((float) setting('profit_percent', 11)) / 100;
 $rounding = (int) setting('price_rounding', 5000);

 $hargaProdukIdr = $hargaRmb * $rate;
 $lokalCn = $this->calculateLocalCn($product, $variant);
 $cnBtm = $this->calculateCnBtm($weightGram);
 $btmJkt = $this->calculateBtmJkt($weightGram);
 $ongkirTotal = $lokalCn + $cnBtm + $btmJkt;

 $modalItem = $hargaProdukIdr + $ongkirTotal;
 $profitIdr = $modalItem * $profitPct;
 $rawTotal = $modalItem + $profitIdr;

 // Pembulatan CEILING ke kelipatan 5.000 per item (Harga Bersih)
 $basePriceBersih = (float) (ceil($rawTotal / $rounding) * $rounding);
 $discountAmount = $product->hasActiveDiscount() ? (float) $product->discount_amount : 0.0;
 $finalPriceBersih = max(0, $basePriceBersih - $discountAmount);

 return [
 'price_rmb' => round($hargaRmb, 2),
 'harga_produk_idr' => round($hargaProdukIdr, 2),
 'weight_gram' => $weightGram,
 'store_name' => $product->effectiveStore()?->name,
 'ongkir_lokal_cn_idr' => round($lokalCn, 2),
 'ongkir_cn_btm_idr' => round($cnBtm, 2),
 'ongkir_btm_jkt_idr' => round($btmJkt, 2),
 'ongkir_total_idr' => round($ongkirTotal, 2),
 'profit_idr' => round($profitIdr, 2),
 'base_price_bersih' => $basePriceBersih,
 'discount_amount' => $discountAmount,
 'final_price_bersih' => $finalPriceBersih,
 ];
 }
}

 /**
 * Total checkout (v1.5). Harga item SUDAH final (dibulatkan sekali di calculateItemShipping),
 * jadi di sini TIDAK ada CEILING lagi. Fee QRIS/marketplace dihitung sesuai pilihan yang aktif
 * dan boleh diganti sebelum/sesudah "Buat Pesanan".
 */
 public function calculateCheckoutTotal(
 array $items, // [['final_price_bersih' => float, 'qty' => int], ...]
 float $discount = 0, // Koin (maks 5% subtotal) ATAU Voucher
 string $scheme = 'fp', // 'fp' | 'dp'
 string $platform = 'toco', // 'toco' | 'shopee'
 string $method = 'bca' // 'bca' | 'qris' | 'shopeepay'
 ): array {
 $subtotal = (float) collect($items)->sum(fn ($i) => $i['final_price_bersih'] * $i['qty']);
 $totalBarang = max(0, $subtotal - $discount);
 $feeQris = ($method === 'qris' && $totalBarang >= 500000) ? $totalBarang * 0.0003 : 0;

 if ($scheme === 'dp') {
 $dp = $totalBarang * 0.5;
 $sisa = $totalBarang - $dp;
 return [
 'subtotal' => $subtotal,
 'discount' => $discount,
 'total_barang' => $totalBarang,
 'fee_qris' => $feeQris,

17.2 Middleware EnforceIpBlacklist (Smart Scoped IP + Device Ban)

app/Http/Middleware/EnforceIpBlacklist.php

<?php

namespace App\Http\Middleware;

use App\Helpers\fungsi IP helper;
use App\Models\SpammerIp;
use Closure;
use Illuminate\Http\Request;

class EnforceIpBlacklist
{
 public function handle(Request $request, Closure $next)
 {
 // 1. Bot Crawler resmi bypass IP Blacklist (untuk SEO)
 if ($request->attributes->get('is_verified_bot', false)) {
 return $next($request);
 }

 // 2. Jika user sudah login sebagai Customer aktif yang sah (MVP / Ex / ONS), izinkan akses penuh
 $user = $request->user();
 if ($user && $user->role === 'customer' && in_array($user->customer?->current_state, ['MVP', 'Ex', 'ONS'], true)) {
 return $next($request);
 }

 // 3. Cek apakah IP masuk daftar spammer_ips yang masih aktif (TTL 30 hari belum lewat)
 $ip = fungsi IP helper::get($request);
 $isBannedIp = SpammerIp::where('ip_address', $ip)
 ->where(function ($q) {
 $q->whereNull('expires_at')->orWhere('expires_at', '>', now());
 })
 ->exists();

 if ($isBannedIp) {
 // Izinkan Beranda ('/') dan halaman '/login' (agar Customer sah di jaringan CGNAT tetap bisa login)
 // Namun blokir total '/register' dan seluruh halaman katalog publik bagi Guest dari IP tersebut
 if ($request->is('/') || $request->is('login') || $request->is('logout')) {
 return $next($request);
 }

 return redirect('/')->with('error', 'Akses pendaftaran dan katalog dibatasi dari jaringan ini.');
 }

 return $next($request);
 }
}

Catatan v1.5 — aturan /login dari IP ter-ban: middleware ini sengaja membiarkan halaman /login terbuka dari IP di spammer_ips. Pengamanannya ada di Layer 1: LoginController menolak email/No HP yang ada di spammers atau Blocked permanen dari IP mana pun. Jadi akun Customer sah (MVP/Ex/ONS) tetap bisa masuk, sedangkan identitas ter-block tetap ditolak.

 /**
 * Simpan backup original di private storage & hasilkan gambar publik ber-watermark permanen.
 * Walaupun orang mencoba screenshot atau ambil dari Inspect Element, gambar sudah ber-watermark!
 */
 public function storeWithWatermark(UploadedFile $file, string $publicRelativePath): string
 {

 // jadi path cukup 'originals/...' (BUKAN 'private/originals/...')
 Storage::disk('local')->put(

 file_get_contents($file->getRealPath())
 );

 $image = $this->manager->read($file->getRealPath());

 // 2. Jika fitur watermark aktif, tanamkan watermark diagonal berulang (tiled)
 if ((bool) setting('image_watermark_enabled', true)) {
 $text = (string) config('mercatoria.image_protection.watermark_text', 'MERCATORIA · mercatoria.id');
 $width = $image->width();
 $height = $image->height();
 $stepX = max(220, (int) ($width / 3));
 $stepY = max(180, (int) ($height / 4));

 for ($y = 60; $y < $height; $y += $stepY) {
 for ($x = 40; $x < $width; $x += $stepX) {
 $image->text($text, $x, $y, function ($font) {
 $font->filename(resource_path('fonts/Inter-Bold.ttf')); // wajib di driver GD agar size() & angle() berlaku
 $font->size(22);
 $font->color('rgba(255, 255, 255, 0.28)');
 $font->stroke('rgba(0, 0, 0, 0.18)', 1);
 $font->angle(-28);
 });
 }
 }
 }

 // 3. Simpan sebagai WebP/JPG ke storage/app/public/{path}
 $encoded = $image->toWebp(85);
 Storage::disk('public')->put($publicRelativePath, (string) $encoded);

 return $publicRelativePath;
 }
}

<?php

namespace App\Services\Split;

use App\Enums\OrderStatus;
use App\Enums\SplitRole;
use App\Exceptions\SlotUnavailableException;
use App\Models\Cart;
use App\Models\Order;
use App\Models\SplitBox;
use App\Models\SplitBoxBatch;
use App\Models\SplitBoxBatchSlot;
use App\Services\Notification\AdminNotificationService;
use App\Services\Notification\UserNotificationService;
use App\Services\Order\OrderStatusService;
use App\Services\Payment\RefundService;
use Illuminate\Support\Facades\DB;

/**
 * ASUMSI API service yang sudah ada (sesuaikan dengan kode asli):
 * - OrderStatusService::transition(Order, OrderStatus, array $context = [])
 * - UserNotificationService::notify(User|int, string $type, string $subtype, string $title, string $message, ?string $url)
 * - AdminNotificationService::notify(string $type, string $message, ?string $url)
 * - RefundService::requestFull(Order, string $reason)
 */
class SplitBoxService
{
 public function __construct(
 protected OrderStatusService $statusService,
 protected UserNotificationService $userNotif,
 protected AdminNotificationService $adminNotif,
 protected RefundService $refunds,
 ) {}

 // ───────────────────────── 1. Aturan pair kumulatif ─────────────────────────

 /** @param iterable<array{role:string,qty:int}> $lines */
 public function evaluatePair(iterable $lines): array
 {
 $normal = 0;
 $hot = 0;
 $required = 0;

 foreach ($lines as $line) {
 $role = SplitRole::from($line['role']);
 $qty = (int) $line['qty'];

 if ($role === SplitRole::Normal) {
 $normal += $qty;
 } else {
 $hot += $qty;
 $required += $role->pairRequirement() * $qty;
 }
 }

 return [
 'normal' => $normal,
 'hot' => $hot,
 'required' => $required,
 'missing' => max(0, $required - $normal),
 'valid' => ($normal + $hot) > 0 && $normal >= $required,
 ];
 }

 // ───────────────────────── 2. Ketersediaan (gabungan kloter terbuka) ─────────────────────────

 public function availableQty(int $slotId): int
 {
 return (int) SplitBoxBatchSlot::query()
 ->where('split_box_slot_id', $slotId)
 ->whereHas('batch', fn ($q) => $q->where('status', 'open'))
 ->selectRaw('COALESCE(SUM(qty_total - qty_taken), 0) AS qty')
 ->value('qty');
 }

 // ───────────────────────── 3. Kunci slot saat BUKTI BAYAR diupload ─────────────────────────

 /**
 * Dipanggil SINKRON dari listener PaymentUploaded, sebelum status berubah ke Ditahan.
 * All-or-nothing: bila satu slot gagal, seluruh transaksi di-rollback dan exception dilempar.
 *
 * @throws SlotUnavailableException
 */
 public function lockForOrder(Order $order): void
 {
 $boxIds = [];

 DB::transaction(function () use ($order, &$boxIds) {
 $items = $order->items()->whereNotNull('split_box_slot_id')->with('splitSlot')->get();
 if ($items->isEmpty()) {
 return;
 }

 $boxIds = $items->map(fn ($i) => $i->splitSlot->split_box_id)->unique()->sort()->values()->all();

 // Row lock berurutan (id naik) agar dua upload bersamaan tidak deadlock
 SplitBox::whereIn('id', $boxIds)->orderBy('id')->lockForUpdate()->get();

 foreach ($items->groupBy('bundle_group_id') as $bundle) {
 $eval = $this->evaluatePair($bundle->map(fn ($i) => [
 'role' => $i->splitSlot->split_role,
 'qty' => $i->qty,
 ]));
 if (! $eval['valid']) {
 throw new SlotUnavailableException('Syarat pair tidak terpenuhi.');
 }
 }

 foreach ($items as $item) {
 if ($item->split_box_batch_slot_id) {
 continue; // sudah terkunci (mis. upload ulang)
 }

 // FIFO lintas kloter: kloter terbuka dengan batch_no terkecil yang masih cukup
 $batchSlot = SplitBoxBatchSlot::query()
 ->where('split_box_slot_id', $item->split_box_slot_id)
 ->whereRaw('qty_taken + ? <= qty_total', [$item->qty])
 ->whereHas('batch', fn ($q) => $q->where('status', 'open'))
 ->orderBy(
 SplitBoxBatch::select('batch_no')
 ->whereColumn('split_box_batches.id', 'split_box_batch_slots.split_box_batch_id')
 )
 ->lockForUpdate()
 ->first();

 if (! $batchSlot) {
 throw new SlotUnavailableException('Slot karakter sudah diambil pembeli lain.');
 }

 $batchSlot->increment('qty_taken', $item->qty);
 $item->update(['split_box_batch_slot_id' => $batchSlot->id]);
 }
 });

 if ($boxIds) {
 DB::afterCommit(fn () => $this->afterSlotsLocked($order, $boxIds));
 }
 }

 /** Lepas slot: bukti ditolak, Pembayaran Gagal, atau Dana Dikembalikan. */
 public function releaseForOrder(Order $order): void
 {
 DB::transaction(function () use ($order) {
 $touched = [];

 foreach ($order->items()->whereNotNull('split_box_batch_slot_id')->get() as $item) {
 $bs = SplitBoxBatchSlot::whereKey($item->split_box_batch_slot_id)->lockForUpdate()->first();
 if ($bs) {
 $bs->update(['qty_taken' => DB::raw('GREATEST(qty_taken - ' . (int) $item->qty . ', 0)')]);
 $touched[$bs->split_box_batch_id] = true;
 }
 $item->update(['split_box_batch_slot_id' => null]);
 }

 SplitBoxBatch::whereIn('id', array_keys($touched))->get()->each(fn ($b) => $this->refreshBatchStatus($b));
 });
 }

 // ───────────────────────── 4. Efek setelah slot terkunci ─────────────────────────

 protected function afterSlotsLocked(Order $order, array $boxIds): void
 {
 foreach ($boxIds as $boxId) {
 $box = SplitBox::with(['slots', 'product'])->find($boxId);
 if (! $box) {
 continue;
 }
 $this->ensureBatchHealth($box); // bisa membuka kloter baru dulu
 $this->purgeUnsatisfiableCarts($box, $order->user_id); // lalu bersihkan keranjang user lain
 $this->cancelUnsatisfiablePendingOrders($box, $order->id);
 }
 }

 protected function purgeUnsatisfiableCarts(SplitBox $box, int $exceptUserId): void
 {
 $slotIds = $box->slots->pluck('id');

 Cart::query()
 ->whereNotNull('bundle_group_id')
 ->where('user_id', '!=', $exceptUserId)
 ->whereIn('split_box_slot_id', $slotIds)
 ->get()
 ->groupBy('bundle_group_id')
 ->each(function ($lines, $group) {

 public function ensureBatchHealth(SplitBox $box): void
 {
 DB::transaction(function () use ($box) {
 $box = SplitBox::whereKey($box->id)->lockForUpdate()->first()->load('slots', 'product');

 $box->batches()->where('status', 'open')->get()->each(fn ($b) => $this->refreshBatchStatus($b));

 $remaining = [];
 foreach ($box->slots as $slot) {
 $remaining[] = ['role' => $slot->role(), 'qty' => $this->availableQty($slot->id)];
 }
 $remaining = collect($remaining);

 $normal = $remaining->filter(fn ($r) => $r['role'] === SplitRole::Normal)->sum('qty');
 $hotNeeds = $remaining
 ->filter(fn ($r) => $r['role'] !== SplitRole::Normal && $r['qty'] > 0)
 ->map(fn ($r) => $r['role']->pairRequirement());

 $soldOut = $remaining->sum('qty') === 0;
 $deadlock = $hotNeeds->isNotEmpty() && $normal < $hotNeeds->min();

 if ($soldOut || $deadlock) {
 $this->openNextBatch($box);
 }
 });
 }

 protected function refreshBatchStatus(SplitBoxBatch $batch): void
 {
 if (! in_array($batch->status, ['open', 'full'], true)) {
 return;
 }
 $remaining = (int) $batch->batchSlots()->selectRaw('COALESCE(SUM(qty_total - qty_taken), 0) AS q')->value('q');
 $batch->update(['status' => $remaining === 0 ? 'full' : 'open']);
 }

 protected function openNextBatch(SplitBox $box): ?SplitBoxBatch
 {
 $autoCount = $box->batches()->where('is_auto_opened', true)->count();

 if (! $box->auto_open_batch || $autoCount >= $box->max_auto_batches) {
 $this->adminNotif->notify(
 'split_box_limit',
 "Split Box {$box->product->name}: kloter otomatis sudah mencapai batas ({$box->max_auto_batches}). Tentukan langkah berikutnya.",
 "/office/products/{$box->product_id}/edit#split-box"
 );
 return null;
 }

 $batch = $box->batches()->create([
 'batch_no' => ((int) $box->batches()->max('batch_no')) + 1,
 'status' => 'open',
 'is_auto_opened' => true,
 'opened_at' => now(),
 ]);

 foreach ($box->slots as $slot) {
 $batch->batchSlots()->create([
 'split_box_slot_id' => $slot->id,
 'qty_total' => $slot->qty_per_box,
 'qty_taken' => 0,
 ]);
 }

 $this->adminNotif->notify(
 'split_box_batch_opened',
 "Kloter {$batch->batch_no} Split Box {$box->product->name} dibuka otomatis. Siapkan pembelian box baru.",
 "/office/products/{$box->product_id}/edit#split-box"
 );

 return $batch;
 }

<?php

namespace App\Services\User;

use App\Models\Customer;
use App\Models\Order;
use Illuminate\Support\Facades\DB;

/**
 * Progress Tier dihitung dari nilai PENUH orders.total (walau DP 50%),
 * hanya ketika status mencapai Sedang Diproses. Pembayaran denda TIDAK dihitung.
 */
class UserTierService
{
 /** Idempotent: aman dipanggil berulang. */
 public function recordProcessing(Order $order): void
 {
 DB::transaction(function () use ($order) {
 $order = Order::whereKey($order->id)->lockForUpdate()->firstOrFail();
 if ($order->tier_counted_at !== null) {
 return;
 }

 $amount = (float) $order->total;
 $customer = Customer::where('user_id', $order->user_id)->lockForUpdate()->firstOrFail();

 $customer->total_belanja = (float) $customer->total_belanja + $amount;
 $order->forceFill(['tier_counted_at' => now(), 'tier_counted_amount' => $amount])->save();

 $this->applyTier($customer);
 });
 }

 /** Full refund / Hangus: $partialAmount = null. Partial refund: nominal yang di-refund. */
 public function reverse(Order $order, ?float $partialAmount = null): void
 {
 DB::transaction(function () use ($order, $partialAmount) {
 $order = Order::whereKey($order->id)->lockForUpdate()->firstOrFail();
 $counted = (float) $order->tier_counted_amount;
 if ($order->tier_counted_at === null || $counted <= 0) {
 return;
 }

 $delta = $partialAmount === null ? $counted : min($partialAmount, $counted);
 $customer = Customer::where('user_id', $order->user_id)->lockForUpdate()->firstOrFail();

 $customer->total_belanja = max(0, (float) $customer->total_belanja - $delta);
 $remaining = $counted - $delta;
 $order->forceFill([
 'tier_counted_amount' => $remaining,
 'tier_counted_at' => $remaining > 0 ? $order->tier_counted_at : null,
 ])->save();

 $this->applyTier($customer);
 });
 }

 /** Denda ditebus: nilai order kembali masuk (denda sendiri tidak ikut). */
 public function reapplyAfterPenalty(Order $order): void
 {
 $this->recordProcessing($order);
 }

 public function tierFor(float $total): ?string
 {
 return match (true) {
 $total > (float) setting('tier_platinum_threshold', 5_000_000) => 'platinum',
 $total > (float) setting('tier_gold_threshold', 2_000_000) => 'gold',
 $total > 0 => 'silver',
 default => null,
 };
 }

 protected function applyTier(Customer $customer): void
 {
 $customer->tier = $this->tierFor((float) $customer->total_belanja);
 $customer->save();
 }

 /** Data untuk progress bar Header Dashboard. */
 public function progress(Customer $customer): array
 {
 $total = (float) $customer->total_belanja;
 $gold = (float) setting('tier_gold_threshold', 2_000_000);
 $platinum = (float) setting('tier_platinum_threshold', 5_000_000);
 $tier = $this->tierFor($total);

 return match ($tier) {
 null => ['tier' => null, 'next' => 'silver', 'percent' => 0, 'remaining' => null, 'total' => $total],
 'silver' => ['tier' => 'silver', 'next' => 'gold', 'percent' => min(100, $total / $gold * 100),
 'remaining' => max(0, $gold - $total) + 1, 'total' => $total],
 'gold' => ['tier' => 'gold', 'next' => 'platinum', 'percent' => min(100, ($total - $gold) / ($platinum - $gold) * 100),
 'remaining' => max(0, $platinum - $total) + 1, 'total' => $total],
 default => ['tier' => 'platinum', 'next' => null, 'percent' => 100, 'remaining' => 0, 'total' => $total],
 };
 }
}

class LatePenaltyService
{
 public function __construct(
 protected OrderStatusService $statusService,
 protected UserTierService $tier,
 protected UserNotificationService $userNotif,
 ) {}

 public function accruedDays(Order $order): int
 {
 return max(1, (int) ceil(abs($order->hangus_at->floatDiffInDays(now()))));
 }

 public function quote(Order $order): array
 {
 $perDay = (float) setting('late_penalty_per_day', 1000);
 $days = $this->accruedDays($order);

 return ['per_day' => $perDay, 'days' => $days, 'amount' => $perDay * $days];
 }

 public function unlock(Order $order, User $admin, ?float $overrideAmount = null): Order
 {
 abort_unless($order->status === OrderStatus::Hangus, 422, 'Hanya order berstatus Hangus yang bisa dibuka kuncinya.');

 $q = $this->quote($order);

 DB::transaction(function () use ($order, $admin, $q, $overrideAmount) {
 $order->forceFill([
 'penalty_per_day' => $q['per_day'],
 'penalty_days' => $q['days'],
 'penalty_amount' => $overrideAmount ?? $q['amount'],
 'penalty_unlocked_at' => now(),
 'penalty_deadline_at' => now()->addHours((int) setting('late_penalty_pay_hours', 24)),
 'penalty_paid_at' => null,
 ])->save();

 $this->statusService->transition($order, OrderStatus::MenungguDenda, ['admin_id' => $admin->id]);
 });

 $this->userNotif->notify(

 );

 return $order->refresh();
 }

 public function confirmPaid(Order $order): void
 {

 $order->forceFill([
 'penalty_paid_at' => now(),
 'pickup_deadline_at' => now()->addDays((int) setting('late_penalty_pickup_days', 7)),
 ])->save();

 $this->statusService->transition($order, OrderStatus::MenungguPelunasanPengambilan, ['via' => 'penalty']);
 $this->tier->reapplyAfterPenalty($order); // denda sendiri tidak dihitung ke tier
 });

 $this->userNotif->notify(
 $order->user_id, 'order', 'late_penalty_paid', 'Denda terverifikasi',

 );
 }

 public function expireUnpaid(): void
 {

 ->where('status', OrderStatus::MenungguDenda->value)
 ->where('penalty_deadline_at', '<', now())
 ->get()
 ->each(fn (Order $o) => $this->statusService->transition($o, OrderStatus::Hangus, ['reason' => 'penalty_unpaid']));

 $max = (int) setting('late_penalty_max_days', 0);
 if ($max > 0) {

 ->filter(fn (Order $o) => $this->accruedDays($o) > $max)

 }

 {

 default => 'Syarat pair terpenuhi. Paket siap ditambahkan ke keranjang.',
 };

 }

 {
 $slot = $this->slots[$slotId] ?? null;
 if (! $slot || $slot['available'] < 1) {
 return;
 }

 if (isset($this->selected[$slotId])) {
 unset($this->selected[$slotId]);
 } else {
 $this->selected[$slotId] = 1;
 }
 }

 public function setQty(int $slotId, int $qty): void
 {
 $slot = $this->slots[$slotId] ?? null;
 if (! $slot || ! isset($this->selected[$slotId])) {
 return;
 }
 $this->selected[$slotId] = max(1, min($qty, $slot['max_qty'], $slot['available']));
 }

 public function addBundleToCart(): void
 {
 abort_unless(auth()->check(), 403);

 if (! $this->pair['valid']) {
 $this->addError('pair', $this->pair['message']);
 return;
 }

 $group = (string) Str::uuid();

 $slot = $this->product->splitBox->slots->firstWhere('id', $slotId);

 'split_box_slot_id' => $slotId,
 ]);
 }

 $this->selected = [];
 $this->dispatch('cart-updated');
 $this->dispatch('toast', type: 'success', message: 'Paket Split Box masuk keranjang.');

 public function render()
 {
