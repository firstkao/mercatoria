7. Model BARU: Store NEW

app/Models/Store.php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Store extends Model
{
    protected $fillable = [
        'name',
        'min_belanja_rmb',
        'ongkir_rmb',
    ];

    protected $casts = [
        'min_belanja_rmb' => 'decimal:2',
        'ongkir_rmb' => 'decimal:2',
    ];

    // ─── Relationships ───────────────────────────────────

    public function games(): HasMany
    {
        return $this->hasMany(Game::class, 'default_store_id');
    }

    public function products(): HasMany
    {
        return $this->hasMany(Product::class);
    }

    // ─── Helpers ─────────────────────────────────────────

    /**
     * Cek apakah store gratis selalu (ongkir = 0)
     */
    public function isFreeShipping(): bool
    {
        return $this->ongkir_rmb == 0;
    }

    /**
     * Cek apakah min_belanja aktif (bukan 0)
     */
    public function hasMinBelanja(): bool
    {
        return $this->min_belanja_rmb > 0;
    }
}

9. Service BARU: StoreService NEW

app/Services/Product/StoreService.php
<?php

namespace App\Services\Product;

use App\Models\Store;
use Illuminate\Support\Collection;

class StoreService
{
    /**
     * Get all stores untuk dropdown admin
     */
    public function getAllForDropdown(): Collection
    {
        return Store::orderBy('name')
            ->get(['id', 'name', 'min_belanja_rmb', 'ongkir_rmb']);
    }

    /**
     * Create store baru
     */
    public function create(array $data): Store
    {
        return Store::create([
            'name' => $data['name'],
            'min_belanja_rmb' => $data['min_belanja_rmb'] ?? 0,
            'ongkir_rmb' => $data['ongkir_rmb'] ?? 0,
        ]);
    }

    /**
     * Update store
     */
    public function update(Store $store, array $data): Store
    {
        $store->update([
            'name' => $data['name'] ?? $store->name,
            'min_belanja_rmb' => $data['min_belanja_rmb'] ?? $store->min_belanja_rmb,
            'ongkir_rmb' => $data['ongkir_rmb'] ?? $store->ongkir_rmb,
        ]);

        return $store->fresh();
    }

    /**
     * Hapus store. Produk & game yang pakai jadi NULL.
     */
    public function delete(Store $store): void
    {
        $store->delete();
        // FK nullOnDelete → otomatis products.store_id, games.default_store_id jadi NULL
    }

    /**
     * Count produk & game yang pakai store ini
     */
    public function getUsageCount(Store $store): array
    {
        return [
            'products' => $store->products()->count(),
            'games' => $store->games()->count(),
        ];
    }
}

10. Service BARU: ShippingCalculatorService NEW

app/Services/Order/ShippingCalculatorService.php
<?php

namespace App\Services\Order;

use App\Models\Product;
use App\Models\ProductVariant;

class ShippingCalculatorService
{
    /**
     * Hitung ongkir lokal CN (per produk) dalam IDR
     */
    public function calculateLocalCn(Product $product): float
    {
        $store = $product->effectiveStore();

        // Kalau gak ada store → gratis
        if (! $store) {
            return 0;
        }

        // Kalau ongkir = 0 → gratis selalu
        if ($store->isFreeShipping()) {
            return 0;
        }

        // Kalau min_belanja = 0 → gratis (biar gak divide by zero)
        if (! $store->hasMinBelanja()) {
            return 0;
        }

        $hargaRmb = (float) $product->price_rmb;
        $minBelanja = (float) $store->min_belanja_rmb;
        $ongkirRmb = (float) $store->ongkir_rmb;

        // Kalau harga ≥ min belanja → gratis
        if ($hargaRmb >= $minBelanja) {
            return 0;
        }

        // Hitung: (harga × ongkir) / min_belanja
        $ongkirLokalRmb = ($hargaRmb * $ongkirRmb) / $minBelanja;
        $rate = (float) setting('rate_rmb_idr', 2200);

        return $ongkirLokalRmb * $rate;
    }

    /**
     * Hitung ongkir CN-BTM dalam IDR berdasarkan berat
     */
    public function calculateCnBtm(int $weightGram): float
    {
        $rate = (float) setting('ongkir_cn_btm_per_kg', 25000);
        $beratKg = $weightGram / 1000;

        return $beratKg * $rate;
    }

    /**
     * Hitung ongkir BTM-JKT dalam IDR berdasarkan berat
     */
    public function calculateBtmJkt(int $weightGram): float
    {
        $rate = (float) setting('ongkir_btm_jkt_per_kg', 60000);
        $beratKg = $weightGram / 1000;

        return $beratKg * $rate;
    }

    /**
     * Hitung SEMUA ongkir untuk 1 item
     * Return: array breakdown + total
     */
    public function calculateItemShipping(
        Product $product,
        ?ProductVariant $variant = null
    ): array {
        // Berat efektif: varian dulu, fallback produk
        $weightGram = $variant?->effectiveWeight() ?? $product->effectiveWeight();

        $lokalCn = $this->calculateLocalCn($product);
        $cnBtm = $this->calculateCnBtm($weightGram);
        $btmJkt = $this->calculateBtmJkt($weightGram);

        return [
            'weight_gram' => $weightGram,
            'store_name' => $product->effectiveStore()?->name,
            'ongkir_lokal_cn_idr' => round($lokalCn, 2),
            'ongkir_cn_btm_idr' => round($cnBtm, 2),
            'ongkir_btm_jkt_idr' => round($btmJkt, 2),
            'ongkir_total_idr' => round($lokalCn + $cnBtm + $btmJkt, 2),
        ];
    }
}

11. Service UPDATE: OrderItemSnapshotService UPD

app/Services/Order/OrderItemSnapshotService.php (REVISI — tambah shipping breakdown)
class OrderItemSnapshotService
{
    public function __construct(
        protected ShippingCalculatorService $shippingCalc
    ) {}

    public function createForOrderItem(OrderItem $item, string $orderNo): OrderItemSnapshot
    {
        if (! $item->product || ! $item->variant) {
            throw new \RuntimeException(
                "Produk atau varian tidak ditemukan untuk order item #{$item->id}"
            );
        }

        $product = $item->product;
        $variant = $item->variant;

        // Hitung ongkir breakdown via service
        $shipping = $this->shippingCalc->calculateItemShipping($product, $variant);

        // Developer dari relasi
        $game = $product->games->first();
        $developerName = $game?->developers->first()?->name;

        // Copy gambar
        $productImagePath = $this->copyImage(
            $product->main_image_path,
            "order-snapshots/{$orderNo}/p-{$item->id}.jpg"
        );

        $variantImagePath = null;
        if ($variant->image_path) {
            $variantImagePath = $this->copyImage(
                $variant->image_path,
                "order-snapshots/{$orderNo}/v-{$item->id}.jpg"
            );
        }

        // Harga
        $basePrice = $variant->price > 0
            ? $variant->price
            : $product->hargaJual();

        $finalPrice = $variant->price > 0
            ? $variant->price
            : $product->hargaSetelahDiskon();

        return OrderItemSnapshot::create([
            'order_item_id' => $item->id,
            'product_id_audit' => $product->id,
            'variant_id_audit' => $variant->id,

            // Produk
            'product_name' => $product->name,
            'product_slug' => $product->slug,
            'product_sku' => $product->sku,
            'game_name' => $game?->name,
            'developer_name' => $developerName,
            'store_name' => $shipping['store_name'],  // ⭐ BARU
            'tag' => $product->tags->first()?->name,

            // Varian
            'variant_name' => $variant->name,
            'variant_sku' => $variant->sku,
            'variant_attributes' => $variant->attributesArray(),

            // Harga
            'base_price' => $basePrice,
            'discount_amount' => $product->hasActiveDiscount()
                ? $product->discount_amount
                : 0,
            'final_price' => $finalPrice,
            'qty' => $item->qty,
            'subtotal' => $finalPrice * $item->qty,

            // Berat & ongkir breakdown (v1.2)
            'weight_gram' => $shipping['weight_gram'],
            'ongkir_lokal_cn_idr' => $shipping['ongkir_lokal_cn_idr'],
            'ongkir_cn_btm_idr' => $shipping['ongkir_cn_btm_idr'],
            'ongkir_btm_jkt_idr' => $shipping['ongkir_btm_jkt_idr'],
            'ongkir_total_idr' => $shipping['ongkir_total_idr'],

            // Gambar
            'product_image_path' => $productImagePath,
            'variant_image_path' => $variantImagePath,

            'snapshot_at' => now(),
        ]);
    }

    // ... copyImage() tetap sama seperti v1.1 ...
}
